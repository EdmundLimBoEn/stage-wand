cmake_minimum_required(VERSION 3.21)
project(StageWand LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
add_executable(stagewand main.cpp window.h)
target_link_libraries(stagewand PRIVATE Qt6::Widgets Qt6::DBus)
install(TARGETS stagewand RUNTIME DESTINATION bin)
install(FILES systems.edmundlim.StageWand.desktop DESTINATION share/applications)
install(FILES stagewand.svg DESTINATION share/icons/hicolor/scalable/apps)

include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(stagewand-desktop-test test.cpp window.h)
  target_link_libraries(stagewand-desktop-test PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test)
  add_test(NAME desktop-lifecycle COMMAND stagewand-desktop-test)
  set_tests_properties(desktop-lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
